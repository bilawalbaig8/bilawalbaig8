const fs=require('fs'),path=require('path'),sharp=require('./profile-render-tools/node_modules/sharp'),icons=require('./profile-render-tools/node_modules/simple-icons');
const root=path.join(__dirname,'github-profile'),assets=path.join(root,'assets');
const input=process.argv[2];if(!input)throw new Error('Pass the generated meditation character cutout path.');
const duration=6,frames=72;
const original=fs.readFileSync(path.join(assets,'toolkit-v2.svg'),'utf8');
const match=/<defs><clipPath id="meditation-art">[\s\S]*?<\/g>/;
if(!match.test(original))throw new Error('Expected existing meditation panel not found.');
const W=2*Math.PI/duration;
icons.siOpenai={path:fs.readFileSync(path.join(__dirname,'reference-profile/openai.svg'),'utf8').match(/<path d="([^"]+)"/)[1]};
function token(name,symbol,c,x,y,t,index){const dx=Math.sin(t*W+index*.9)*3,dy=Math.sin(t*W+index*.9)*7,angle=Math.sin(t*W+index)*3;const ic=icons['si'+name];let mark=ic?`<g transform="translate(-13 -19) scale(1.08)" fill="${c}"><path d="${ic.path}"/></g>`:`<text x="0" y="6" text-anchor="middle" font-size="24" fill="${c}" font-weight="700">${symbol}</text>`;
return `<g transform="translate(${x+dx} ${y+dy}) rotate(${angle})"><rect x="-28" y="-29" width="56" height="58" rx="12" fill="#12172b" stroke="${c}" stroke-width="1.6"/><rect x="-30" y="-31" width="60" height="62" rx="14" fill="none" stroke="${c}" stroke-opacity=".12" stroke-width="5"/>${mark}${ic||name==='Openai'?`<text x="0" y="20" text-anchor="middle" font-size="8.5" fill="#d5dbea">${symbol}</text>`:''}</g>`;}
async function main(){
 const cutout=await sharp(input).trim({threshold:4}).resize({width:650,height:650,fit:'inside'}).png({palette:true,quality:95}).toBuffer();
 fs.writeFileSync(path.join(assets,'meditation-character-layer.png'),cutout);
 const data='data:image/png;base64,'+cutout.toString('base64');
 function scene(t){const bob=Math.sin(t*W)*8;let s=`<defs><clipPath id="local-scene"><rect x="25" y="30" width="390" height="430" rx="18"/></clipPath><radialGradient id="local-glow"><stop stop-color="#954eab" stop-opacity=".44"/><stop offset="1" stop-color="#171427" stop-opacity="0"/></radialGradient></defs><g clip-path="url(#local-scene)"><rect x="25" y="30" width="390" height="430" rx="18" fill="#111728"/><ellipse cx="220" cy="252" rx="205" ry="215" fill="url(#local-glow)"/>`;
 // Quiet studio silhouettes give depth without moving the camera or cropping the art.
 s+='<path d="M28 302H412M36 300V202L123 208V292M306 290V208L404 197V300" fill="none" stroke="#7765a2" stroke-opacity=".25" stroke-width="3"/><path d="M47 213L111 217V260H47ZM319 218L393 210V261H319Z" fill="#5879bb" opacity=".09"/>';
 for(let i=0;i<3;i++)s+=`<ellipse cx="220" cy="${426+i*4}" rx="${125+i*9}" ry="${13+i*3}" fill="none" stroke="${i===1?'#b481ff':'#ffb66e'}" opacity="${.32+.16*Math.sin(t*W+i)}" stroke-width="${i===1?2:1}"/>`;
 for(let i=0;i<3;i++)s+=`<circle cx="220" cy="225" r="${74+i*8}" fill="none" stroke="${i===1?'#e5a35d':'#ab82ef'}" opacity="${.15+.05*Math.sin(t*W)}" stroke-width="1"/>`;
 const items=[['Claude','Claude','#f0a178',75,115],['Openai','ChatGPT','#77d5e2',171,77],['Googlegemini','Gemini','#b89afa',269,83],['Deepseek','DeepSeek','#73acff',362,127],['Unity','Unity','#dfebff',65,227],['Blender','Blender','#ffad65',65,327],['Autodeskmaya','M','#6dd1d6',366,235],['Adobephotoshop','Ps','#7bbcff',366,332]];
 for(let i=0;i<items.length;i++){const [name,symbol,c,x,y]=items[i];const ty=y+Math.sin(t*W+i*.9)*7;s+=`<path d="M${x} ${ty}Q220 ${175+Math.sin(t*W)*5}220 271" fill="none" stroke="${c}" stroke-width="1" opacity=".13"/>`;}
 s+=`<image href="${data}" x="95" y="${166+bob}" width="252" height="246" preserveAspectRatio="xMidYMid meet"/>`;
 items.forEach(([name,symbol,c,x,y],i)=>s+=token(name,symbol,c,x,y,t,i));
 for(let i=0;i<15;i++){const a=i*2.4+t*W,r=88+i%4*18;s+=`<circle cx="${220+Math.cos(a)*r}" cy="${270+Math.sin(a)*r*.8}" r="${i%3===0?1.6:1}" fill="${i%2?'#c19aff':'#ffd48b'}" opacity=".6"/>`;}
 return s+'</g>';}
 const render=t=>original.replace(match,scene(t));
 const staticSvg=render(0);fs.writeFileSync(path.join(assets,'toolkit-floating.svg'),staticSvg);
 await sharp(Buffer.from(staticSvg)).png().toFile(path.join(assets,'toolkit-floating.png'));
 const dir=path.join(__dirname,'profile-frames','toolkit-floating');fs.mkdirSync(dir,{recursive:true});
 for(let i=0;i<frames;i++)await sharp(Buffer.from(render(i/12))).resize(1100).png().toFile(path.join(dir,String(i).padStart(3,'0')+'.png'));
 console.log('Rendered full-body character and eight independently floating tokens; 72 local frames, no crop.');
}
main().catch(e=>{console.error(e);process.exitCode=1});
